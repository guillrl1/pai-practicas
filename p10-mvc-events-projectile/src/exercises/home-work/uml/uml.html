    %% Model layer
    class Projectile {
        -GRAVITY: number$
        -initialVelocity: number
        -angleDegrees: number
        -initialHeight: number
        +getPositionAt(time: number) ProjectileState
        +getTotalFlightTime() number
        +getMaxHeight() number
        -toRadians(deg: number) number
    }
    class ProjectileState {
        <<interface>>
        +x: number
        +y: number
        +time: number
    }

    %% View layer
    class CanvasView {
        <<abstract>>
        #canvas: HTMLCanvasElement
        #ctx: CanvasRenderingContext2D
        #cssWidth: number
        #cssHeight: number
        +resize() void
        +clear() void
    }
    class MainView {
        -scale: number
        -margin: number
        -worldRangeX: number
        -worldRangeY: number
        -scaleFixed: boolean
        +resize() void
        +setWorldRange(rx: number, ry: number) void
        +drawProjectile(x: number, y: number, color: string) void
        +drawCannon(h0: number, ang: number) void
        +clearAll() void
        +resetScale() void
        -recalculateScale() void
        -drawAxes() void
    }
    class InfoView {
        -maxObservedHeight: number
        +resetMaxHeight() void
        +renderStats(t: number, x: number, y: number) void
        +clear() void
    }
    class ControlView {
        -animateButton: HTMLButtonElement
        -inputVelocity: HTMLInputElement
        -inputAngle: HTMLInputElement
        -inputHeight: HTMLInputElement
        -checkboxPath: HTMLInputElement
        +getSimulationParams() Object
        +onAnimate(callback: Function) void
    }

    %% Controller layer
    class ProjectileController {
        -projectile: Projectile
        -isAnimating: boolean
        -startTime: number
        -animFrameId: number
        -colors: string[]
        +startSimulation() void
        -stopAnimation() void
        -animationLoop(ts: number) void
        -setupListeners() void
    }

    %% Relationships
    CanvasView <|-- MainView : Inheritance
    CanvasView <|-- InfoView : Inheritance
    CanvasView <|-- ControlView : Inheritance
    ProjectileController o-- MainView : Aggregation
    ProjectileController o-- InfoView : Aggregation
    ProjectileController o-- ControlView : Aggregation
    ProjectileController ..> Projectile : Instantiates
    Projectile ..> ProjectileState : Produces
Componente Responsabilidad en el Sistema
Modelo (Projectile) Cálculos físicos puros. Independiente del navegador.
Vistas (CanvasView) Gestión del elemento HTML5 Canvas y soporte HiDPI/Resize.
Controlador Orquesta la comunicación y gestiona el bucle requestAnimationFrame.
